```python
"""
Regra R050 — Ativo = Passivo + PL (equação patrimonial).

Dimensão: Interconta
Severidade padrão: ERRO

Lógica:
    Soma dos saldos finais líquidos da classe 1 (Ativo) deve igualar
    a soma dos saldos finais líquidos da classe 2 (Passivo + PL).

    ativo   = SUM(SF_D - SF_C)  para contas onde conta_contabil[0] == "1"
    passivo = SUM(SF_C - SF_D)  para contas onde conta_contabil[0] == "2"
    SE abs(ativo - passivo) > tolerancia → ERRO

Referência:
    Portaria STN 642/2026 — Regras Gerais MSC, R050
"""

from __future__ import annotations

import logging
from typing import TYPE_CHECKING

import pandas as pd

from ranking_msc.rules.base import TOLERANCIA_CENTAVO, Rule, register_rule

if TYPE_CHECKING:
    from ranking_msc.engine.executor import EngineContext

logger = logging.getLogger("ranking_msc")


@register_rule
class R050(Rule):
    """Verifica que o Ativo (classe 1) = Passivo + PL (classe 2)."""

    rule_id = "R050"
    rule_name = "Ativo = Passivo + PL"
    dimensao = "Interconta"

    def execute(self, df: pd.DataFrame, context: EngineContext) -> pd.DataFrame:
        """Detecta desequilíbrio entre Ativo (classe 1) e Passivo+PL (classe 2).

        Args:
            df: DataFrame normalizado com colunas conta_contabil,
                saldo_final_debito e saldo_final_credito.
            context: metadados da execução.

        Returns:
            DataFrame de pendências — vazio se equilíbrio, uma linha se desequilíbrio.
        """
        _extra = {"regra_id": self.rule_id, "competencia": context.competencia}

        if df.empty:
            return self.build_issues([])

        work = df.copy()
        for col in ("saldo_final_debito", "saldo_final_credito"):
            work[col] = pd.to_numeric(work[col], errors="coerce").fillna(0.0)

        classe1 = work[work["conta_contabil"].str[:1] == "1"]
        classe2 = work[work["conta_contabil"].str[:1] == "2"]

        ativo = (classe1["saldo_final_debito"] - classe1["saldo_final_credito"]).sum()
        passivo = (classe2["saldo_final_credito"] - classe2["saldo_final_debito"]).sum()

        diferenca = abs(ativo - passivo)

        if diferenca <= TOLERANCIA_CENTAVO:
            return self.build_issues([])

        logger.warning(
            "R050: Ativo (%.2f) ≠ Passivo+PL (%.2f), diferença=%.2f. competência=%s",
            ativo,
            passivo,
            diferenca,
            context.competencia,
            extra=_extra,
        )

        return self.build_issues([{
            "severidade": "ERRO",
            "conta_contabil": "",
            "descricao": (
                f"Equação patrimonial desequilibrada: "
                f"Ativo={ativo:.2f}, Passivo+PL={passivo:.2f}, "
                f"diferença={diferenca:.2f}."
            ),
            "valor_esperado": f"Ativo={ativo:.2f}",
            "valor_encontrado": f"Passivo+PL={passivo:.2f}",
            "competencia": context.competencia,
        }])
```

```python
"""
Regra R051 — VPA − VPD = Resultado do Exercício.

Dimensão: Interconta
Severidade padrão: ALERTA

Lógica:
    O resultado operacional (VPA − VPD) deve ser consistente com a
    variação do grupo Resultados Acumulados (PCASP 2.1.4, prefixo "214").

    vpa_liq   = SUM(mov_C - mov_D)  para contas onde conta_contabil[0] == "4"
    vpd_liq   = SUM(mov_D - mov_C)  para contas onde conta_contabil[0] == "3"
    resultado = vpa_liq - vpd_liq

    delta_pl  = SUM(SF_C - SF_D) - SUM(SI_C - SI_D)
                para contas onde conta_contabil[:3] == "214"

    SE abs(resultado - delta_pl) > tolerancia → ALERTA

Limitação conhecida:
    Ajustes diretos no PL que não transitam por classes 3/4 (doações,
    correções monetárias) podem gerar falso ALERTA. Isso é esperado —
    por isso a severidade é ALERTA e não ERRO.

Referência:
    Portaria STN 642/2026 — Regras Gerais MSC, R051
"""

from __future__ import annotations

import logging
from typing import TYPE_CHECKING

import pandas as pd

from ranking_msc.rules.base import TOLERANCIA_CENTAVO, Rule, register_rule

if TYPE_CHECKING:
    from ranking_msc.engine.executor import EngineContext

logger = logging.getLogger("ranking_msc")


@register_rule
class R051(Rule):
    """Verifica consistência entre VPA-VPD e variação dos Resultados Acumulados (2.1.4)."""

    rule_id = "R051"
    rule_name = "VPA - VPD = Resultado"
    dimensao = "Interconta"

    def execute(self, df: pd.DataFrame, context: EngineContext) -> pd.DataFrame:
        """Compara resultado operacional com variação do grupo 2.1.4 (Resultados Acumulados).

        Args:
            df: DataFrame normalizado com colunas conta_contabil, movimento_debito,
                movimento_credito, saldo_inicial_debito, saldo_inicial_credito,
                saldo_final_debito, saldo_final_credito.
            context: metadados da execução.

        Returns:
            DataFrame de pendências — vazio se consistente ou sem dados 214;
            uma linha se divergência detectada.
        """
        _extra = {"regra_id": self.rule_id, "competencia": context.competencia}

        if df.empty:
            return self.build_issues([])

        cols_valor = [
            "movimento_debito", "movimento_credito",
            "saldo_inicial_debito", "saldo_inicial_credito",
            "saldo_final_debito", "saldo_final_credito",
        ]
        work = df.copy()
        for col in cols_valor:
            work[col] = pd.to_numeric(work[col], errors="coerce").fillna(0.0)

        grupo214 = work[work["conta_contabil"].str[:3] == "214"]

        if grupo214.empty:
            logger.info(
                "R051: nenhuma conta do grupo 2.1.4 encontrada — validação ignorada. competência=%s",
                context.competencia,
                extra=_extra,
            )
            return self.build_issues([])

        classe3 = work[work["conta_contabil"].str[:1] == "3"]
        classe4 = work[work["conta_contabil"].str[:1] == "4"]

        vpa_liq = (classe4["movimento_credito"] - classe4["movimento_debito"]).sum()
        vpd_liq = (classe3["movimento_debito"] - classe3["movimento_credito"]).sum()
        resultado = vpa_liq - vpd_liq

        pl_sf = (grupo214["saldo_final_credito"] - grupo214["saldo_final_debito"]).sum()
        pl_si = (grupo214["saldo_inicial_credito"] - grupo214["saldo_inicial_debito"]).sum()
        delta_pl = pl_sf - pl_si

        diferenca = abs(resultado - delta_pl)

        if diferenca <= TOLERANCIA_CENTAVO:
            return self.build_issues([])

        logger.warning(
            "R051: VPA-VPD (%.2f) ≠ delta PL 2.1.4 (%.2f), diferença=%.2f. competência=%s",
            resultado,
            delta_pl,
            diferenca,
            context.competencia,
            extra=_extra,
        )

        return self.build_issues([{
            "severidade": "ALERTA",
            "conta_contabil": "",
            "descricao": (
                f"Resultado operacional inconsistente com Resultados Acumulados (2.1.4): "
                f"VPA-VPD={resultado:.2f}, ΔPL(2.1.4)={delta_pl:.2f}, "
                f"diferença={diferenca:.2f}."
            ),
            "valor_esperado": f"VPA-VPD={resultado:.2f}",
            "valor_encontrado": f"delta_PL(2.1.4)={delta_pl:.2f}",
            "competencia": context.competencia,
        }])
```

```python
"""
Regra R052 — Controles devedores = Controles credores (7/8).

Dimensão: Interconta
Severidade padrão: ALERTA

Lógica:
    Saldos finais líquidos da classe 7 (Controles Devedores) devem
    igualar saldos finais líquidos da classe 8 (Controles Credores).

    ctrl_d = SUM(SF_D - SF_C)  para contas onde conta_contabil[0] == "7"
    ctrl_c = SUM(SF_C - SF_D)  para contas onde conta_contabil[0] == "8"
    SE abs(ctrl_d - ctrl_c) > tolerancia → ALERTA

Referência:
    Portaria STN 642/2026 — Regras Gerais MSC, R052
"""

from __future__ import annotations

import logging
from typing import TYPE_CHECKING

import pandas as pd

from ranking_msc.rules.base import TOLERANCIA_CENTAVO, Rule, register_rule

if TYPE_CHECKING:
    from ranking_msc.engine.executor import EngineContext

logger = logging.getLogger("ranking_msc")


@register_rule
class R052(Rule):
    """Verifica que os saldos finais da classe 7 igualam os da classe 8."""

    rule_id = "R052"
    rule_name = "Controles devedores = Controles credores"
    dimensao = "Interconta"

    def execute(self, df: pd.DataFrame, context: EngineContext) -> pd.DataFrame:
        """Detecta desequilíbrio entre controles devedores (7) e credores (8).

        Args:
            df: DataFrame normalizado com colunas conta_contabil,
                saldo_final_debito e saldo_final_credito.
            context: metadados da execução.

        Returns:
            DataFrame de pendências — vazio se equilíbrio, uma linha se desequilíbrio.
        """
        _extra = {"regra_id": self.rule_id, "competencia": context.competencia}

        if df.empty:
            return self.build_issues([])

        work = df.copy()
        for col in ("saldo_final_debito", "saldo_final_credito"):
            work[col] = pd.to_numeric(work[col], errors="coerce").fillna(0.0)

        classe7 = work[work["conta_contabil"].str[:1] == "7"]
        classe8 = work[work["conta_contabil"].str[:1] == "8"]

        if classe7.empty and classe8.empty:
            return self.build_issues([])

        ctrl_d = (classe7["saldo_final_debito"] - classe7["saldo_final_credito"]).sum()
        ctrl_c = (classe8["saldo_final_credito"] - classe8["saldo_final_debito"]).sum()

        diferenca = abs(ctrl_d - ctrl_c)

        if diferenca <= TOLERANCIA_CENTAVO:
            return self.build_issues([])

        logger.warning(
            "R052: Controles 7 (%.2f) ≠ Controles 8 (%.2f), diferença=%.2f. competência=%s",
            ctrl_d,
            ctrl_c,
            diferenca,
            context.competencia,
            extra=_extra,
        )

        return self.build_issues([{
            "severidade": "ALERTA",
            "conta_contabil": "",
            "descricao": (
                f"Desequilíbrio entre controles: "
                f"Classe 7={ctrl_d:.2f}, Classe 8={ctrl_c:.2f}, "
                f"diferença={diferenca:.2f}."
            ),
            "valor_esperado": f"Classe 7={ctrl_d:.2f}",
            "valor_encontrado": f"Classe 8={ctrl_c:.2f}",
            "competencia": context.competencia,
        }])
```
