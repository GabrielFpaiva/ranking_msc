```python
"""
ranking_msc.benchmark.generator
~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

Gerador de CSVs MSC sintéticos para testes de carga.

Produz arquivos no mesmo formato dos reais de ``ref/amostras/``:
    - Linha 1: ``<siconfi>;<competencia>`` (metadata)
    - Linha 2: cabeçalho de colunas
    - Linhas 3+: dados
    - Line terminator: ``\\r`` (CR only)
    - Separador: ``;``
    - Encoding: ASCII

As combinações IC/TIPO são sorteadas de ``ref/mapeamento_ic_tipo.csv`` para
garantir que o arquivo gerado passe por ``read_msc_csv`` + ``normalize_ic``
sem levantar exceções.

Não busca realismo estatístico — é stress test, não simulação.
"""

from __future__ import annotations

import logging
from pathlib import Path

import numpy as np
import pandas as pd

from ranking_msc.exceptions import BenchmarkError

logger = logging.getLogger("ranking_msc")

_SYNTHETIC_SICONFI: str = "9999999EX"
_SYNTHETIC_COMPETENCIA: str = "2026-01"

_CSV_HEADER: str = (
    "CONTA;IC1;TIPO1;IC2;TIPO2;IC3;TIPO3;IC4;TIPO4;"
    "IC5;TIPO5;IC6;TIPO6;Valor;Tipo_valor;Natureza_valor"
)

_TIPO_VALOR_DOMAIN: tuple[str, ...] = (
    "beginning_balance", "period_change", "ending_balance",
)
_NATUREZA_VALOR_DOMAIN: tuple[str, ...] = ("D", "C")

_DEFAULT_MAPPING_PATH: Path = (
    Path(__file__).resolve().parents[3] / "ref" / "mapeamento_ic_tipo.csv"
)
_DEFAULT_PCASP_PATH: Path = (
    Path(__file__).resolve().parents[3] / "ref" / "pcasp.csv"
)


def generate_csv(
    n_rows: int,
    path: Path,
    *,
    seed: int = 42,
    mapping_path: Path = _DEFAULT_MAPPING_PATH,
    pcasp_path: Path = _DEFAULT_PCASP_PATH,
) -> Path:
    """Gera um CSV MSC sintético com ``n_rows`` linhas de dados.

    Args:
        n_rows: quantidade de linhas de dados a gerar (sem contar metadata/header).
        path: caminho do arquivo a criar (diretórios pais são criados).
        seed: semente para reprodutibilidade.
        mapping_path: caminho de ``ref/mapeamento_ic_tipo.csv``.
        pcasp_path: caminho de ``ref/pcasp.csv``.

    Returns:
        ``path`` passado como argumento.

    Raises:
        BenchmarkError: se arquivos de referência estiverem ausentes.
    """
    if n_rows <= 0:
        raise BenchmarkError(f"n_rows deve ser positivo, recebido: {n_rows}")

    path.parent.mkdir(parents=True, exist_ok=True)

    rng = np.random.default_rng(seed)
    contas = _load_contas_analiticas(pcasp_path)
    combinations = _load_ic_tipo_combinations(mapping_path)

    conta_escolhas = rng.choice(contas, size=n_rows)
    combo_idx = rng.integers(0, len(combinations), size=n_rows)
    valores = rng.uniform(-1e7, 1e7, size=n_rows)
    tipos_valor_idx = rng.integers(0, len(_TIPO_VALOR_DOMAIN), size=n_rows)
    naturezas_idx = rng.integers(0, len(_NATUREZA_VALOR_DOMAIN), size=n_rows)

    lines: list[str] = [f"{_SYNTHETIC_SICONFI};{_SYNTHETIC_COMPETENCIA}", _CSV_HEADER]
    for i in range(n_rows):
        ics, tipos = combinations[combo_idx[i]]
        conta = conta_escolhas[i]
        valor = f"{valores[i]:.2f}"
        tipo_valor = _TIPO_VALOR_DOMAIN[tipos_valor_idx[i]]
        natureza = _NATUREZA_VALOR_DOMAIN[naturezas_idx[i]]
        fields = [str(conta)]
        for j in range(6):
            fields.append(ics[j])
            fields.append(tipos[j])
        fields.extend([valor, tipo_valor, natureza])
        lines.append(";".join(fields))

    content = "\r".join(lines) + "\r"
    path.write_bytes(content.encode("ascii"))

    logger.info(
        "CSV MSC sintético gerado: %d linhas em %s (seed=%d)",
        n_rows, path, seed,
        extra={"modulo": "benchmark.generator"},
    )
    return path


def _load_contas_analiticas(pcasp_path: Path) -> list[str]:
    """Carrega contas PCASP analíticas (nivel != 'Superior')."""
    if not pcasp_path.exists():
        raise BenchmarkError(f"Arquivo PCASP não encontrado: {pcasp_path}")

    try:
        df = pd.read_csv(pcasp_path, sep=";", dtype=str, keep_default_na=False)
    except Exception as exc:  # noqa: BLE001
        raise BenchmarkError(f"Falha ao ler PCASP em {pcasp_path}: {exc}") from exc

    required = {"conta_contabil", "nivel"}
    missing = required - set(df.columns)
    if missing:
        raise BenchmarkError(
            f"PCASP sem colunas obrigatórias {sorted(missing)}. Arquivo: {pcasp_path}"
        )

    nivel = df["nivel"].str.strip()
    analiticas = df.loc[nivel != "Superior", "conta_contabil"].tolist()
    if not analiticas:
        raise BenchmarkError(f"PCASP sem contas analíticas utilizáveis: {pcasp_path}")
    return analiticas


def _load_ic_tipo_combinations(
    mapping_path: Path,
) -> list[tuple[tuple[str, str, str, str, str, str], tuple[str, str, str, str, str, str]]]:
    """Carrega combinações válidas de TIPO1..TIPO6 com ICs plausíveis.

    Retorna uma lista de tuplas ``((ic1..ic6), (tipo1..tipo6))``. Para cada TIPO
    preenchido, o IC correspondente é um código placeholder "1".
    ICs/TIPOs vazios no mapeamento ficam como string vazia.
    """
    if not mapping_path.exists():
        raise BenchmarkError(f"Mapeamento IC/TIPO não encontrado: {mapping_path}")

    try:
        df = pd.read_csv(mapping_path, sep=";", dtype=str, keep_default_na=False)
    except Exception as exc:  # noqa: BLE001
        raise BenchmarkError(
            f"Falha ao ler mapeamento IC/TIPO em {mapping_path}: {exc}"
        ) from exc

    tipo_cols = [f"tipo{i}" for i in range(1, 7)]
    missing = [c for c in tipo_cols if c not in df.columns]
    if missing:
        raise BenchmarkError(
            f"Mapeamento IC/TIPO sem colunas {missing}. Arquivo: {mapping_path}"
        )

    combinations: list[tuple[tuple[str, ...], tuple[str, ...]]] = []
    for _, row in df.iterrows():
        tipos = tuple(str(row[c]).strip().upper() for c in tipo_cols)
        ics = tuple("1" if t else "" for t in tipos)
        combinations.append((ics, tipos))

    if not combinations:
        raise BenchmarkError(
            f"Mapeamento IC/TIPO vazio (sem linhas): {mapping_path}"
        )
    return combinations
```

"""
ranking_msc.benchmark.profiler
~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

Instrumentação de performance: mede wall time e uso de memória por etapa
do pipeline e por regra individual.

APIs principais:
    - ``measure(name)``: context manager para medir uma etapa.
    - ``instrument_rules()``: context manager que captura timing por regra
      (patcha ``Rule.execute`` temporariamente).
    - ``BenchmarkResult`` / ``StageMetrics`` / ``RuleMetrics``: dataclasses
      de retorno.

Nota sobre memória: ``resource.getrusage`` reporta RSS máximo do processo
(global), não por etapa. Usamos RSS delta por etapa como aproximação
(útil pra identificar picos) e peak global como número honesto no
``BenchmarkResult``.
"""

from __future__ import annotations

import logging
import resource
import sys
import time
from contextlib import contextmanager
from dataclasses import dataclass, field
from typing import Iterator

logger = logging.getLogger("ranking_msc")


@dataclass(frozen=True)
class StageMetrics:
    """Métricas de uma etapa do pipeline."""

    name: str
    wall_seconds: float
    rss_delta_mb: float


@dataclass(frozen=True)
class RuleMetrics:
    """Métricas de execução de uma regra individual."""

    rule_id: str
    wall_seconds: float
    issue_count: int
    failed: bool = False


@dataclass(frozen=True)
class BenchmarkResult:
    """Resultado consolidado de uma rodada de benchmark pra um tamanho."""

    n_rows: int
    total_seconds: float
    peak_rss_mb: float
    throughput_rows_per_sec: float
    stages: list[StageMetrics] = field(default_factory=list)
    rules: list[RuleMetrics] = field(default_factory=list)
    failed: bool = False
    failure_reason: str | None = None


# ---------------------------------------------------------------------------
# Context manager: measure
# ---------------------------------------------------------------------------

@dataclass
class _MeasureAccumulator:
    """Objeto mutável retornado por ``measure()`` dentro do bloco with.

    Após sair do bloco ``with``, os campos estão populados. Use
    ``.to_stage()`` se precisar imobilizar em ``StageMetrics``.
    """

    name: str
    wall_seconds: float = 0.0
    rss_delta_mb: float = 0.0

    def to_stage(self) -> StageMetrics:
        return StageMetrics(
            name=self.name,
            wall_seconds=self.wall_seconds,
            rss_delta_mb=self.rss_delta_mb,
        )


def _get_rss_mb() -> float:
    """Retorna RSS atual do processo em MB.

    ``ru_maxrss`` é reportado em KB no Linux e em bytes no macOS.
    """
    rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    if sys.platform == "darwin":
        return rss / (1024 * 1024)
    return rss / 1024


@contextmanager
def measure(name: str) -> Iterator[_MeasureAccumulator]:
    """Mede wall time e RSS delta do bloco ``with``.

    Uso:
        with measure("read_csv") as m:
            do_work()
        print(m.wall_seconds, m.rss_delta_mb)
    """
    acc = _MeasureAccumulator(name=name)
    rss_before = _get_rss_mb()
    t0 = time.perf_counter()
    try:
        yield acc
    finally:
        acc.wall_seconds = time.perf_counter() - t0
        acc.rss_delta_mb = _get_rss_mb() - rss_before
```

# ---------------------------------------------------------------------------
# Context manager: instrument_rules
# ---------------------------------------------------------------------------


@contextmanager
def instrument_rules() -> Iterator[list[RuleMetrics]]:
    """Instrumenta ``Rule.execute`` de cada regra registrada.

    Dentro do bloco ``with``, cada chamada a ``rule.execute(df, context)``
    é cronometrada. A lista de ``RuleMetrics`` é popular conforme regras
    executam. Ao sair do bloco, os patches são revertidos.

    Uso:
        with instrument_rules() as metrics:
            run_engine(df, ctx)
        # metrics agora tem RuleMetrics de cada regra executada

    Só patcha regras que já estão em ``RULE_REGISTRY`` no momento do
    ``__enter__``.
    """
    from ranking_msc.rules.base import RULE_REGISTRY

    metrics: list[RuleMetrics] = []
    originals: dict[str, object] = {}

    def _make_wrapper(rule_id: str, original):  # type: ignore[no-untyped-def]
        def wrapper(self, df, context):  # type: ignore[no-untyped-def]
            t0 = time.perf_counter()
            failed = False
            issue_count = 0
            try:
                out = original(self, df, context)
                try:
                    issue_count = len(out)
                except TypeError:
                    issue_count = 0
                return out
            except Exception:
                failed = True
                raise
            finally:
                metrics.append(RuleMetrics(
                    rule_id=rule_id,
                    wall_seconds=time.perf_counter() - t0,
                    issue_count=issue_count,
                    failed=failed,
                ))
        return wrapper

    try:
        for rule_id, rule_cls in RULE_REGISTRY.items():
            originals[rule_id] = rule_cls.execute
            rule_cls.execute = _make_wrapper(rule_id, rule_cls.execute)  # type: ignore[method-assign]
        yield metrics
    finally:
        for rule_id, original in originals.items():
            RULE_REGISTRY[rule_id].execute = original  # type: ignore[method-assign]
```

Nota: o import do `RULE_REGISTRY` é feito dentro da função pra evitar que o import do profiler force o import das regras ainda no momento de carregar o módulo.

"""
ranking_msc.benchmark.reporter
~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

Geração do relatório markdown consolidado a partir de uma lista de
``BenchmarkResult``.

Estrutura gerada:
    - Header com data, commit, versões de Python/pandas.
    - Seção "Resumo geral": 1 linha por tamanho.
    - Seção "Tempo por etapa": 1 linha por tamanho, colunas por etapa.
    - Seção "Top-5 regras mais lentas": 1 subseção por tamanho.

Se um resultado tem ``failed=True``, a linha aparece com "FAIL: <razão>".
"""

from __future__ import annotations

import logging
import platform
import subprocess
import sys
from datetime import datetime
from pathlib import Path

import pandas as pd

from ranking_msc.benchmark.profiler import BenchmarkResult

logger = logging.getLogger("ranking_msc")


def write_report(results: list[BenchmarkResult], output_path: Path) -> None:
    """Escreve o relatório markdown em ``output_path``.

    Args:
        results: lista de BenchmarkResult, um por tamanho testado.
        output_path: caminho de saída (diretórios pais são criados).
    """
    output_path.parent.mkdir(parents=True, exist_ok=True)

    lines: list[str] = []
    lines.extend(_header())
    lines.append("")
    lines.extend(_secao_resumo_geral(results))
    lines.append("")
    lines.extend(_secao_tempo_por_etapa(results))
    lines.append("")
    lines.extend(_secao_top_regras(results))

    output_path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    logger.info("Relatório de benchmark escrito em %s", output_path)


def _header() -> list[str]:
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    commit = _git_commit()
    py_ver = ".".join(str(x) for x in sys.version_info[:3])
    pd_ver = pd.__version__
    machine = platform.processor() or platform.machine()

    return [
        "# Benchmark de Performance — ranking_msc",
        "",
        f"**Gerado em:** {now}",
        f"**Commit:** `{commit}`",
        f"**Python:** {py_ver} · **pandas:** {pd_ver}",
        f"**Máquina:** {machine}",
    ]


def _git_commit() -> str:
    try:
        out = subprocess.check_output(
            ["git", "rev-parse", "--short", "HEAD"],
            stderr=subprocess.DEVNULL,
            text=True,
        )
        return out.strip()
    except (subprocess.CalledProcessError, FileNotFoundError):
        return "unknown"


def _fmt_seconds(s: float) -> str:
    if s < 1:
        return f"{s * 1000:.0f}ms"
    if s < 60:
        return f"{s:.1f}s"
    m, s_ = divmod(s, 60)
    return f"{int(m)}m{s_:.0f}s"


def _fmt_mb(mb: float) -> str:
    if mb < 1024:
        return f"{mb:.0f} MB"
    return f"{mb / 1024:.2f} GB"


def _fmt_throughput(tps: float) -> str:
    if tps < 1000:
        return f"{tps:.0f} rows/s"
    return f"{tps / 1000:.1f}k rows/s"


def _secao_resumo_geral(results: list[BenchmarkResult]) -> list[str]:
    lines = [
        "## Resumo geral",
        "",
        "| Tamanho | Tempo total | Memória pico | Throughput |",
        "|---|---|---|---|",
    ]
    for r in results:
        if r.failed:
            lines.append(f"| {r.n_rows} | FAIL: {r.failure_reason} | — | — |")
        else:
            lines.append(
                f"| {r.n_rows} | {_fmt_seconds(r.total_seconds)} | "
                f"{_fmt_mb(r.peak_rss_mb)} | {_fmt_throughput(r.throughput_rows_per_sec)} |"
            )
    return lines


def _secao_tempo_por_etapa(results: list[BenchmarkResult]) -> list[str]:
    lines = [
        "## Tempo por etapa",
        "",
        "| Tamanho | read_csv | normalize_ic | run_engine |",
        "|---|---|---|---|",
    ]
    for r in results:
        if r.failed:
            lines.append(f"| {r.n_rows} | FAIL | FAIL | FAIL |")
            continue
        stages = {s.name: s for s in r.stages}
        lines.append(
            f"| {r.n_rows} | "
            f"{_fmt_seconds(stages['read_csv'].wall_seconds) if 'read_csv' in stages else '—'} | "
            f"{_fmt_seconds(stages['normalize_ic'].wall_seconds) if 'normalize_ic' in stages else '—'} | "
            f"{_fmt_seconds(stages['run_engine'].wall_seconds) if 'run_engine' in stages else '—'} |"
        )
    return lines


def _secao_top_regras(results: list[BenchmarkResult]) -> list[str]:
    lines = ["## Top-5 regras mais lentas", ""]
    for r in results:
        lines.append(f"### {r.n_rows} linhas")
        lines.append("")
        if r.failed:
            lines.append(f"FAIL: {r.failure_reason}")
            lines.append("")
            continue
        top = sorted(r.rules, key=lambda x: x.wall_seconds, reverse=True)[:5]
        if not top:
            lines.append("_(sem regras medidas)_")
            lines.append("")
            continue
        lines.append("| Regra | Tempo | Pendências |")
        lines.append("|---|---|---|")
        for m in top:
            lines.append(
                f"| {m.rule_id} | {_fmt_seconds(m.wall_seconds)} | {m.issue_count} |"
            )
        lines.append("")
    return lines
```

def _run_one(

    n_rows: int,
    seed: int,
    csv_dir: Path,
    context: EngineContext,

    try:
        generate_csv(n_rows=n_rows, path=csv_path, seed=seed)
    except MemoryError:
        logger.error("MemoryError ao gerar CSV com n_rows=%d", n_rows)
        return BenchmarkResult(
            n_rows=n_rows, total_seconds=0.0, peak_rss_mb=0.0,
            throughput_rows_per_sec=0.0, stages=[], rules=[],

        logger.error("Falha ao gerar CSV com n_rows=%d: %s", n_rows, exc, exc_info=True)
        return BenchmarkResult(
            n_rows=n_rows, total_seconds=0.0, peak_rss_mb=0.0,
            throughput_rows_per_sec=0.0, stages=[], rules=[],
            failed=True, failure_reason=f"Falha na geração: {exc}",
        )

    try:
        with measure("total") as total:
            with measure("read_csv") as m_read:
                raw = read_msc_csv(csv_path)
            with measure("normalize_ic") as m_norm:
                df_norm = normalize_ic(raw.df)
            with instrument_rules() as rule_metrics:
                with measure("run_engine") as m_engine:
                    run_engine(df_norm, context)

        peak = _get_rss_mb()
        return BenchmarkResult(
            n_rows=n_rows,
            total_seconds=total.wall_seconds,
            peak_rss_mb=peak,
            throughput_rows_per_sec=n_rows / total.wall_seconds if total.wall_seconds else 0.0,
            stages=[m_read.to_stage(), m_norm.to_stage(), m_engine.to_stage()],
            rules=list(rule_metrics),
        )
    except MemoryError:
        logger.error("MemoryError no pipeline com n_rows=%d", n_rows)
        return BenchmarkResult(
            n_rows=n_rows, total_seconds=0.0, peak_rss_mb=0.0,
            throughput_rows_per_sec=0.0, stages=[], rules=[],
            failed=True, failure_reason="OOM no pipeline",
        )
    except Exception as exc:  # noqa: BLE001
        logger.error("Falha no pipeline com n_rows=%d: %s", n_rows, exc, exc_info=True)
        return BenchmarkResult(
            n_rows=n_rows, total_seconds=0.0, peak_rss_mb=0.0,
            throughput_rows_per_sec=0.0, stages=[], rules=[],
            failed=True, failure_reason=f"Pipeline falhou: {exc}",
        )


if __name__ == "__main__":
    sys.exit(main())
```
